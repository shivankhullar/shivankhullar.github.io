{{- define "main" -}} {{/* Photography gallery, with two views the visitor can switch between: - Grid: justified rows (every photo in a row has the same height, nothing is cropped), read left to right in the order below; - Scroll: one row of large photos, scrolled left/right (arrows, keys, wheel, swipe). Clicking a photo in either view opens it full screen. The chosen view is remembered. Captions come from `resources:` in content/photography/_index.md: one entry per photo, with `title` (where) and `params.taken` (month and year). With `shuffle: true` there, the photos appear in a new random order on every visit; otherwise in the order of that list (photos not listed are added at the end, by filename). Visitors never download the files in content/photography/ (`_build.publishResources: false`); this template publishes WebP copies instead, at three sizes (800, 1600 and up to 3200 px), and the browser picks the smallest one that is sharp on the visitor's screen. Re-encoding also drops all metadata. */}} {{- $exts := slice "jpg" "jpeg" "png" "webp" "tif" "tiff" -}} {{- $photos := slice -}} {{- $seen := slice -}} {{- range .Params.resources -}} {{- with $.Resources.GetMatch .src -}} {{- if not (in $seen .Name) -}} {{- $photos = $photos | append . -}} {{- $seen = $seen | append .Name -}} {{- end -}} {{- end -}} {{- end -}} {{- range sort (.Resources.ByType "image") "Name" -}} {{- $ext := lower (strings.TrimPrefix "." (path.Ext .Name)) -}} {{- if and (in $exts $ext) (not (in $seen .Name)) (not (hasPrefix (lower (path.Base .Name)) "featured")) -}} {{- $photos = $photos | append . -}} {{- end -}} {{- end -}}
